Google blocks browsers from reading Drive file bytes directly (CORS). The fix below is a tiny proxy that runs on your own Google account — one-time setup, then every Drive link just works.

Set up the proxy (one time, ~2 min)
  1. Go to script.google.com → New project
  2. Delete the placeholder code, paste in the code below
  3. Tap Deploy → New deployment
  4. Type: Web app · Execute as: Me · Who has access: Anyone
  5. Authorize it, then copy the Web app URL (ends in /exec)
  6. Paste that URL into the field below — you only do this once, ever
function doGet(e) {
  var id = e.parameter.id;
  var file = DriveApp.getFileById(id);
  var blob = file.getBlob();
  var b64 = Utilities.base64Encode(blob.getBytes());
  return ContentService
    .createTextOutput(JSON.stringify({
      name: file.getName(),
      mimeType: blob.getContentType(),
      data: b64
    }))
    .setMimeType(ContentService.MimeType.JSON);
}
